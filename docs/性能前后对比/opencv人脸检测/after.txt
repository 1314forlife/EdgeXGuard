// 🟢 优化后：实时流动态跳帧降频推理 + 坐标缓存复用（丝滑 30 帧 + 动态无人脸提示）
// =========================================================================
void FaceCaptureDialog::handleVideoFrame(const QVideoFrame &frame)
{
    if (!frame.isValid()) return;
    m_currentFrame = frame;

    QVideoFrame cloneFrame(frame);
    if (!cloneFrame.map(QVideoFrame::ReadOnly)) return;
    QImage img = cloneFrame.toImage();
    cloneFrame.unmap();
    if (img.isNull()) return;

    // 1. 深拷贝格式对齐，彻底阻断 Qt 临时图层生命周期引发的野指针崩溃
    QImage openCvImage = img.convertToFormat(QImage::Format_RGB888);
    cv::Mat matFrame(openCvImage.height(), openCvImage.width(), CV_8UC3,
                     const_cast<uchar*>(openCvImage.bits()), openCvImage.bytesPerLine());
    cv::cvtColor(matFrame, matFrame, cv::COLOR_RGB2BGR);

    // 2. 🪐【H.264 帧间预测级核心优化】：引入时间冗余消除补丁
    static int frameCounter = 0;
    static std::vector<cv::Rect> cachedFaces;
    frameCounter++;

    if (frameCounter % 4 == 0) { // 🚀 相当于 H.264 的 I帧，每 4 帧才放行执行一次昂贵的算力推理
        cv::Mat gray;
        cv::cvtColor(matFrame, gray, cv::COLOR_BGR2GRAY);
        cv::equalizeHist(gray, gray);

        cv::CascadeClassifier faceCascade;
        if (faceCascade.load("haarcascade_frontalface_alt.xml")) {
            // 调大比例因子至 1.2，过滤 120x120 以下的杂质背景，算力开销再度砍半
            faceCascade.detectMultiScale(gray, cachedFaces, 1.2, 3, 0, cv::Size(120, 120));
        }
        frameCounter = 0; 
    }

    // 3. 🪐【动态状态机分流渲染机制】
    if (!cachedFaces.empty()) {
        // A 状态：命中人脸 ──► 动态绘制绿色加粗追踪框（中间 3 帧直接复用 cached 坐标，零算力开销）
        for (const auto& rect : cachedFaces) {
            cv::rectangle(matFrame, rect, cv::Scalar(0, 255, 0), 3); 
        }
    } else {
        // B 状态：丢失人脸 ──► 物理像素图层强行烫烙半透明红色全屏警告，并拉起引导虚线框
        cv::putText(matFrame, "WARNING: NO FACE DETECTED", cv::Point(30, 50), 
                    cv::FONT_HERSHEY_SIMPLEX, 0.8, cv::Scalar(0, 0, 255), 2, cv::LINE_AA);
        
        int cx = matFrame.cols / 2; int cy = matFrame.rows / 2;
        cv::rectangle(matFrame, cv::Point(cx - 80, cy - 100), cv::Point(cx + 80, cy + 100), cv::Scalar(0, 0, 180), 1, cv::LINE_AA);
    }

    // 4. 洗回 Qt 架构进行 30 帧高速满格渲染
    cv::cvtColor(matFrame, matFrame, cv::COLOR_BGR2RGB);
    QImage resultImage(matFrame.data, matFrame.cols, matFrame.rows, matFrame.step, QImage::Format_RGB888);

    QPixmap pixmap = QPixmap::fromImage(resultImage).scaled(m_videoLabel->size(), Qt::KeepAspectRatioByExpanding, Qt::SmoothTransformation);
    QMetaObject::invokeMethod(m_videoLabel, [this, pixmap]() { m_videoLabel->setPixmap(pixmap); }, Qt::QueuedConnection);
}

// 🟢 优化后：点击抓拍按钮，只负责纯洁的空拦截与落盘写入，与繁重的推理逻辑彻底解耦！
void FaceCaptureDialog::onSaveButtonClicked()
{
    QString name = m_nameEdit->text().trimmed();
    QString workId = m_workIdEdit->text().trimmed();

    if (name.isEmpty() || workId.isEmpty() || !m_currentFrame.isValid()) {
        QMessageBox::warning(this, "警告", "请确保信息填写完整！");
        return;
    }

    QImage capturedImage = m_currentFrame.toImage();
    if (capturedImage.isNull()) return;


    QString folderPath = "faces"; QDir dir;
    if (!dir.exists(folderPath)) dir.mkdir(folderPath);

    QString fileName = QString("%1/%2_%3.jpg").arg(folderPath).arg(workId).arg(QDateTime::currentMSecsSinceEpoch());
    if (!capturedImage.save(fileName)) return;

    FaceDatabaseViewModel::instance().addUser(name, workId, fileName);
    QMessageBox::information(this, "采集成功", QString("用户【%1】白名单凭证已永久安全落库！").arg(name));
    this->accept();
}
